中间件重设计

用统一的 RBAC 中间件替换现有的 requireRole / requireDataPermission

旧 → 新 对照

现有替换为说明
requireRole('admin') requirePermission('admin:access') 路由级权限检查，基于权限码而非硬编码角色名
requireDataPermission('catalog', 'read') requirePermission('catalog:read') 数据操作权限，统一为 perm_code 格式
requireAuth requireAuth（不变） 认证检查不变，只验证登录态
stripInternalIds applyFieldConstraints 按角色权限中的 field_constraints 动态过滤响应字段

新中间件签名

auth.js 新增导出
// 权限检查（替换 requireRole + requireDataPermission）
function requirePermission(permCode) {
  return (req, res, next) => {
    if (!req.session.userId) return send401(res);

    // 1. 查用户所有角色 → 合并权限列表（含继承）
    const perms = getUserEffectivePermissions(req.session.userId);

    // 2. 检查是否拥有所需权限
    if (!perms.has(permCode)) return send403(res, '权限不足');

    // 3. 将有效权限挂载到 req，供后续使用
    req.effectivePermissions = perms;
    next();
  };
}

// 字段级过滤（替换 stripInternalIds）
function applyFieldConstraints(resourceType) {
  return (req, res, next) => {
    const originalJson = res.json.bind(res);
    res.json = function(body) {
      const constraints = getFieldConstraints(
        req.effectivePermissions, resourceType
      );
      return originalJson(applyConstraints(body, constraints));
    };
    next();
  };
}
